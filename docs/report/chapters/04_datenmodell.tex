\section{Datenmodell und Speicherung}

\subsection{MongoDB als Raw Store}
\begin{itemize}
  \item \textbf{Collections:}
    \begin{itemize}
      \item \texttt{insider\_trades\_raw}: Rohe Trade-Dokumente im JSON-Format von API1.
      \item \texttt{companies}: Gecachte Company-Profile von API2.
    \end{itemize}
  \item \textbf{Dokument-Struktur:}
    \begin{itemize}
      \item Jedes \texttt{insider\_trades\_raw}-Dokument enthält die API1-Antwort plus systemische Felder (\texttt{dedupe\_key}, \texttt{fetched\_at}).
      \item Kein Schema-Enforcing -- flexibel bei API-Payload-Änderungen.
      \item Eindeutiger Index auf \texttt{dedupe\_key} verhindert Duplikate. % Repo-Evidenz prüfen
    \end{itemize}
  \item \textbf{Deduplizierung:}
    \begin{itemize}
      \item \texttt{dedupe\_key} = Hash aus (symbol + filing\_date + reporting\_cik + qty + price + direct\_indirect). % Interne Projektspezifikation
      \item Upsert-Semantik: \texttt{\$setOnInsert} aktualisiert nie existierende Dokumente. % Repo-Evidenz prüfen
      \item Mehrfachimporte aus derselben Feed-Seite erzeugen keine Duplikate.
    \end{itemize}
  \item \textbf{Rollen des Raw Store:}
    \begin{itemize}
      \item Audit-Hub für Nachvollziehbarkeit und Debugging.
      \item Reimport-Quelle für Raw $\rightarrow$ Clean-Sync bei Verarbeitungsänderungen.
      \item Keine Geschäftslogik oder Validierung im Raw Store selbst.
    \end{itemize}
\end{itemize}

\subsection{MySQL als Clean Store}
\begin{itemize}
  \item \textbf{Zentrale Tabellen:}
    \begin{itemize}
      \item \textbf{insider\_trades}: Normalisierte Trade-Datensätze (einen pro dedupe\_key).
        \begin{itemize}
          \item Felder: symbol\_at\_trade, filing\_date, transaction\_date, reporting\_cik, company\_cik, company\_key, qty, price, trade\_value, acquisition\_or\_disposition, transaction\_code\_class, validation\_status, gate\_status, score, score\_class, final\_class, decision\_status, dedupe\_key, fetched\_at.
          \item Weitere Felder: market\_cap, industry, sma\_50, sma\_200, momentum\_3m, momentum\_6m, technical\_state, liquidity\_state, core\_insider\_score, investability\_score, execution\_score, trade\_republic\_score, final\_score, filing\_age\_days.
        \end{itemize}
      \item \textbf{companies}: Unternehmensprofile.
        \begin{itemize}
          \item Felder: company\_key, cik, symbol, name, market\_cap, industry, sector, country.
          \item Eindeutiger Schlüssel: company\_key (CIK oder Symbol).
        \end{itemize}
      \item \textbf{market\_signal\_cache}: Technische Metriken pro Symbol und Datum.
        \begin{itemize}
          \item Felder: symbol, date, sma\_50, sma\_200, momentum\_3m, momentum\_6m, avg\_volume\_20d, avg\_dollar\_volume\_20d. % Repo-Evidenz prüfen
        \end{itemize}
      \item \textbf{app\_api\_usage}: FMP-API-Nutzungsstatistiken.
        \begin{itemize}
          \item Felder: date, provider (FMP), call\_count, limit.
        \end{itemize}
      \item \textbf{app\_runtime\_settings, app\_filter\_settings}: Konfigurierbare Parameter.
        \begin{itemize}
          \item Gate-Schwellen, Score-Schwellen, Import-Intervalle.
        \end{itemize}
    \end{itemize}
  \item \textbf{Normalisierung:}
    \begin{itemize}
      \item Companies wird normalisiert (eindeutig pro CIK/Symbol).
      \item insider\_trades wird eindeutig pro dedupe\_key.
      \item Ungültige oder price\_invalid Trades (validation\_status = INVALID / PRICE\_INVALID) werden als reine Audit-Einträge märkiert, nicht exportiert. % Repo-Evidenz prüfen
    \end{itemize}
  \item \textbf{Indizes:}
    \begin{itemize}
      \item PRIMARY KEY: dedupe\_key (insider\_trades), company\_key (companies). % Repo-Evidenz prüfen
      \item INDEX: filing\_date, transaction\_date, symbol\_at\_trade, gate\_status, score, reporting\_cik.
      \item Foreign Key (optional): insider\_trades.company\_key $\rightarrow$ companies.company\_key. % Repo-Evidenz prüfen
    \end{itemize}
\end{itemize}

\subsection{Schlüssel und Deduplizierung}
\begin{itemize}
  \item \textbf{dedupe\_key:} Verhindert Mehrfachimporte derselben API1-Einträge. % Interne Projektspezifikation
    \begin{itemize}
      \item Berechnung: \texttt{hash(symbol + filing\_date + reporting\_cik + qty + price + direct\_indirect)}.
      \item Invariant: Ein dedupe\_key entspricht genau einem wirtschaftlichen Transaktionsereignis.
    \end{itemize}
  \item \textbf{company\_key:} Eindeutige Identität eines Unternehmens (typisch: CIK oder Symbol). % Repo-Evidenz prüfen
  \item \textbf{Upsert-Logik in MySQL:} % Repo-Evidenz prüfen
    \begin{itemize}
      \item \texttt{INSERT ... ON DUPLICATE KEY UPDATE} aktualisiert bestehende Einträge.
      \item Bei jedem Re-Import wird der neueste Stand der Felder geschrieben.
    \end{itemize}
\end{itemize}

\subsection{Rollen der Speicher}
\begin{itemize}
  \item \textbf{Raw Store (MongoDB):}
    \begin{itemize}
      \item Quelle der Wahrheit für unverarbeitete API-Daten.
      \item Ermöglicht Debugging bei Verarbeitungsfehlern.
      \item Unterstützt Reimport-Szenarios.
      \item Keine Abhängigkeit von MySQL-Schema während des Importslaufs.
    \end{itemize}
  \item \textbf{Clean Store (MySQL):}
    \begin{itemize}
      \item Einzige Quelle für alle analytischen Abfragen und UI-Visualisierungen.
      \item Sichere Datenqualität durch Validierung und Normalisierung.
      \item Effiziente Indizierung und Abfrageleistung für interaktive Seiten.
      \item Struktur und Beziehungen sind dokumentiert und wartbar.
    \end{itemize}
  \item \textbf{Synchronisations-Richtung:} MongoDB $\rightarrow$ MySQL (nicht bidirektional). % Interne Projektspezifikation
    \begin{itemize}
      \item Im Importpfad: API1 $\rightarrow$ MongoDB (Raw) $\rightarrow$ (Validierung/Gating/Scoring) $\rightarrow$ MySQL (Clean). % Repo-Evidenz prüfen
      \item \textit{Raw $\rightarrow$ Clean Sync} (optional im Admin): Liest Raw-Daten neu, re-evaluiert Gates und Scores.
    \end{itemize}
\end{itemize}
