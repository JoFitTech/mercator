\section{Datengrundlage}

\subsection{Datenquellen}
\begin{itemize}
  \item \textbf{Primäre Quelle:} Financial Modeling Prep (FMP) API. % Repo-Evidenz prüfen
  \item \textbf{API1 --- Insider Trading Feed:} \texttt{GET /stable/insider-trading/latest} mit Pagination (page, limit). % Repo-Evidenz prüfen
    \begin{itemize}
      \item Liefert die neuesten verfügbaren Insider-Trade-Einträge.
      \item Default-Konfiguration: \texttt{page=0}, \texttt{limit=100}. % Interne Projektspezifikation
      \item Normalisierung erfolgt lokal; Feldmappings werden in \texttt{normalize\_insider\_trade()} durchgeführt. % Repo-Evidenz prüfen
    \end{itemize}
  \item \textbf{API2 --- Company Profile:} \texttt{GET /stable/profile?symbol=\{SYMBOL\}}. % Repo-Evidenz prüfen
    \begin{itemize}
      \item Wird nur für Gate-bestätigte Trades abgerufen (Filterung vor API2). % Interne Projektspezifikation
      \item Liefert Unternehmensmetadaten wie Name, Marktkapitalisierung, Industrie, CIK.
      \item Profile werden in einem TTL-basierten Cache zwischengespeichert.
    \end{itemize}
  \item \textbf{API2 Fallbacks (nur bei Identitätsproblemen):}
    \begin{itemize}
      \item \texttt{GET /stable/search-cik} -- Suche nach CIK bei Symbol-zu-CIK-Aufloesungsproblemen.
      \item \texttt{GET /stable/profile-cik?cik=\{CIK\}} -- Profil direkt per CIK abrufen.
      \item Diese Fallbacks gehören \textbf{nicht} zum Standard-Importpfad. % Interne Projektspezifikation
    \end{itemize}
  \item \textbf{API3 --- Historische Kurs- und Volumendaten:} \texttt{GET /stable/historical-price-eod/full?symbol=\{SYMBOL\}\&from=\{DATE\_FROM\}\&to=\{DATE\_TO\}}. % Repo-Evidenz prüfen
    \begin{itemize}
      \item Wird nach erfolgreichem API2-Abruf für technische Analysen durchgeführt.
      \item Standard-Rückblick: 500 Tage. % Interne Projektspezifikation
      \item Liefert Schluss-, Volumen- und Liquiditätsdaten für technische Metriken (SMA50/200, Momentum).
    \end{itemize}
\end{itemize}

\subsection{Fachlicher Kontext der Insider-Trades}
\begin{itemize}
  \item \textbf{Rechtsrahmen:} Insider-Trades werden in den USA über SEC Form 4 gemeldet. % Externe Quelle nötig
    \begin{itemize}
      \item Eine Gating-Regel: nur \texttt{formType == 4} werden akzeptiert. % Interne Projektspezifikation
    \end{itemize}
  \item \textbf{Transaction Codes als fachliche Klassifikation:}
    \begin{itemize}
      \item \texttt{P} (Purchase) = stärkster positiver Kaufcode. % Interne Projektspezifikation
      \item \texttt{S} (Sale) = stärkster negativer Verkaufscode. % Interne Projektspezifikation
      \item \texttt{I, L} = sekundäre Signale. % Interne Projektspezifikation
      \item \texttt{J, V} = manuelle Überprüfung erforderlich. % Interne Projektspezifikation
      \item \texttt{A, C, D, E, F, G, H, M, O, U, W, X, Z, K} = ausgeschlossen aus Core-Scoring (technische oder entschädigungsnahe Codes). % Interne Projektspezifikation
      \item Vollständige Klassifikation: siehe \texttt{fmp\_api\_spec\_v2\_final.md}. % Repo-Evidenz prüfen
    \end{itemize}
  \item \textbf{Handelbare Instrumente:}
    \begin{itemize}
      \item Nur Aktien und ETFs sind im finalen Kaufsignalmodell zugelassen. % Interne Projektspezifikation
      \item ADRs, Fonds, Optionen und abgeleitete Instrumente sind ausgeschlossen. % Interne Projektspezifikation
      \item Normalisierung erfolgt in \texttt{SecurityNormalizationService}. % Repo-Evidenz prüfen
    \end{itemize}
  \item \textbf{Marktinformationen:}
    \begin{itemize}
      \item Marktkapitalisierung und Sektor aus API2-Profilen.
      \item Präferenz: direkte Symbole gegenüber CIK; CIK-Fallbacks nur bei Problemen.
    \end{itemize}
\end{itemize}

\subsection{Datenformate und Datenstruktur}
\begin{itemize}
  \item \textbf{Raw Store (MongoDB):}
    \begin{itemize}
      \item Speichert rohe, flexible JSON-Payloads von API-Antworten.
      \item Collections: \texttt{insider\_trades\_raw}, \texttt{companies}.
      \item Jedem Raw-Dokument wird ein eindeutiger \texttt{dedupe\_key} zugewiesen (Symbol + Filing Date + Reporting CIK + Qty + Price + Direct/Indirect Hash). % Repo-Evidenz prüfen
      \item Keine Validierung oder Anreicherung im Raw Store.
    \end{itemize}
  \item \textbf{Clean Store (MySQL):}
    \begin{itemize}
      \item Speichert normalisierte, validierte und analysierbare Tabellen.
      \item \textbf{insider\_trades:} normalisierte Trade-Datensätze mit Gating-, Validierungs- und Scoring-Status.
      \item \textbf{companies:} Unternehmensprofile (Name, CIK, Market Cap, Sektor, Industrie).
      \item \textbf{market\_signal\_cache:} Technische Metriken pro Symbol (SMA50/200, Momentum, Liquidität).
      \item \textbf{app\_api\_usage:} Nutzungsstatistiken der FMP-API.
      \item \textbf{app\_filter\_settings, app\_runtime\_settings:} Konfigurierbare Gate- und Score-Schwellen.
      \item Jede Clean-Tabelle wird über fachliche Schlüssel (z. B. \texttt{dedupe\_key}) mit dem Raw Store in Bezug gesetzt. % Repo-Evidenz prüfen
    \end{itemize}
  \item \textbf{Wichtige Felder im Datensatz:}
    \begin{itemize}
      \item \texttt{symbol\_at\_trade}, \texttt{filing\_date}, \texttt{transaction\_date}, \texttt{reporting\_name}, \texttt{company\_cik}, \texttt{reporting\_cik}.
      \item \texttt{qty}, \texttt{price}, \texttt{trade\_value} (geschätzt oder berechnet).
      \item \texttt{transaction\_type}, \texttt{transaction\_code\_class} (normalisiert).
      \item \texttt{validation\_status}, \texttt{gate\_status}, \texttt{score}, \texttt{score\_class}.
      \item \texttt{market\_cap}, \texttt{industry}, \texttt{sma\_50}, \texttt{sma\_200}, \texttt{momentum\_3m} (aus API2/API3).
    \end{itemize}
\end{itemize}
